# Runtime dependencies.
#
# On the ZMB machines the environment is built from conda-forge (see README);
# this file exists so CI and a Linux server can install the same set with pip,
# which is licensing-safe and avoids pulling the Anaconda `defaults` channel.
#
# Bounds are "compatible release": patch and minor updates are accepted, a major
# version bump is not, so a dependency cannot silently change behaviour under a
# facility that only reboots the server twice a year.

fastapi~=0.141
uvicorn~=0.52
sqlalchemy~=2.0

# Schema migrations. `create_all` adds a missing table and never alters an
# existing one, so without this a column added in a release simply never
# reaches a database that already exists.
alembic~=1.18

# The [email] extra pulls in email-validator, which pydantic's EmailStr needs at
# import time. The conda environment happens to get it transitively, so leaving
# it implicit here made the pip install work locally and fail everywhere else.
pydantic[email]~=2.13
pydantic-settings~=2.14
argon2-cffi~=25.1
pillow~=12.3
python-multipart~=0.0.32
python-ulid~=4.0

# Only needed when RETICLE_STORAGE_BACKEND=s3. Kept here rather than in an
# extras group because a facility that switches to object storage should not
# also have to discover a second install command - and an unused wheel costs
# nothing next to Pillow. app/storage.py imports it lazily, so a local-disk
# install never loads it.
boto3~=1.42
